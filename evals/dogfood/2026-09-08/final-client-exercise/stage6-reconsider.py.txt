import json
import runpy
from capture import BASE

session = runpy.run_path(str(BASE / 'notebook-session.py'))
cli, DID = session['cli'], session['DID']
ids = json.loads((BASE / 'note-ids.json').read_text())
original_ids = dict(ids)
ids['command_evidence'] = json.loads((BASE / 'apply-output.json').read_text())['updates'][0]['note_id']
reconsidered = []

def record(name, kind, content, dependencies=(), status=None, source=None):
    current = cli('get', DID, display=False)
    argv = ['record', DID, '--revision', str(current['revision']), '--kind', kind, '--content', content, '--depends-on', json.dumps([ids[key] for key in dependencies])]
    if status:
        argv += ['--status', status]
    if source:
        argv += ['--source', source]
    result = cli(*argv, display=False)
    ids[name] = result['note_id']
    after = cli('get', DID, display=False)
    observed = next(note for note in after['notes'] if note['id'] == result['note_id'])
    reconsidered.append({'name': name, 'previous_note_id': original_ids.get(name), 'new_note': observed, 'revision': after['revision']})
    (BASE / 'reconsideration-readbacks.json').write_text(json.dumps(reconsidered, indent=2) + '\n')
    (BASE / 'current-note-ids.json').write_text(json.dumps(ids, indent=2) + '\n')
    print(json.dumps(reconsidered[-1]))

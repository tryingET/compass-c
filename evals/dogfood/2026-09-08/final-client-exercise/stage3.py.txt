import json
import runpy
from capture import BASE

session = runpy.run_path(str(BASE / 'notebook-session.py'))
cli, DID = session['cli'], session['DID']
ids = {'command_evidence': json.loads((BASE / 'initial-evidence-record.json').read_text())['note_id']}

def record(name, kind, content, dependencies=(), status=None, source=None):
    current = cli('get', DID, display=False)
    argv = ['record', DID, '--revision', str(current['revision']), '--kind', kind, '--content', content, '--depends-on', json.dumps([ids[key] for key in dependencies])]
    if status:
        argv += ['--status', status]
    if source:
        argv += ['--source', source]
    result = cli(*argv, display=False)
    ids[name] = result['note_id']
    (BASE / 'note-ids.json').write_text(json.dumps(ids, indent=2) + '\n')
    readback = cli('get', DID, display=False)
    observed = next(note for note in readback['notes'] if note['id'] == result['note_id'])
    print(json.dumps({'name': name, 'record_result': result, 'readback_note': observed}))

function admit(context, maxTokens, id) {
  const cost = reserve(context, maxTokens, model.cost);
  if (budget.reservedUSD + cost > 10) {
    halted = true;
    throw new Error("Budget reserve exhausted; stop campaign");
  }
  const key = `${id}-${++budget.requests}`;
  budget.reservedUSD += cost;
  budget.inflight[key] = { id, reservedUSD: cost };
  persist();
  return key;
}
function complete(message, key) {
  budget.reportedUSD += message.usage?.cost?.total ?? 0;
  if (!["error", "aborted"].includes(message.stopReason))
    delete budget.inflight[key];
  persist();
}
const options = (maxTokens, temperature) => ({
  maxTokens,
  temperature,
  maxRetries: 0,
  cacheRetention: "none",
  onPayload: (p) => {
    assert.equal(p.max_tokens ?? p.max_completion_tokens, maxTokens);
    budget.storeFalseRequests++;
    persist();
    return { ...p, store: false };
  },
});
function bridge(db, tool, args) {
  const argv = [join(ROOT, "scripts/mcp_call.py"), "--db", db];
  if (tool) argv.push("--tool", tool, "--arguments", JSON.stringify(args));
  return JSON.parse(
    execFileSync(python, argv, {
      encoding: "utf8",
      timeout: 60000,
      maxBuffer: 131072,
    }),
  );
}

async function trial(item) {
  const file = join(output, `${item.id}.json`);
  if (existsSync(file)) {
    assert(["executed", "bounded_failure"].includes(load(file).status));
    return;
  }
  const cwd = join(output, "workspaces", item.id),
    db = join(cwd, "notebook.sqlite3");
  assert(!existsSync(cwd), "Partial trial directory requires reconciliation");
  mkdirSync(cwd, { recursive: true });
  execFileSync("git", ["init", "-q", cwd]);
  if (item.skill)
    execFileSync(python, [
      join(ROOT, "install_skill.py"),
      "--root",
      join(cwd, ".agents/skills"),
    ]);
  const settingsManager = SettingsManager.inMemory({
    packages: [],
    compaction: { enabled: false },
    retry: { enabled: false, provider: { maxRetries: 0 } },
  });
  const loader = new DefaultResourceLoader({
    cwd,
    agentDir: join(home, ".pi/agent"),
    settingsManager,
    noExtensions: true,
    noContextFiles: true,
    noThemes: true,
    noPromptTemplates: true,
    noSkills: !item.skill,
    systemPromptOverride: () => system,
  });
  await loader.reload({ resolveProjectTrust: async () => true });
  assert.deepEqual(loader.getSkills().diagnostics, []);
  assert.deepEqual(
    loader.getSkills().skills.map((x) => x.name),
    item.skill ? ["compass"] : [],
  );
  if (item.skill)
    assert.equal(
      treeHash(join(cwd, ".agents/skills/compass")),
      protocol.skillTreeSha256,
    );
  const customTools =
    item.caseId === "no_python_available"
      ? []
      : discovered.tools.map((t) => ({
          name: t.name,
          label: t.name,
          description: t.description,
          parameters: t.inputSchema,
          execute: async (_id, args) => {
            const r = bridge(db, t.name, args);
            if (r.isError) throw new Error(JSON.stringify(r.content));
            return { content: r.content, details: r };
          },
        }));
  const { session } = await createAgentSession({
    cwd,
    agentDir: join(home, ".pi/agent"),
    modelRuntime: runtime,
    model,
    thinkingLevel: "off",
    resourceLoader: loader,
    settingsManager,
    sessionManager: SessionManager.inMemory(cwd),
    tools: ["read", ...customTools.map((x) => x.name)],
    customTools,
  });
  const messages = [],
    tools = [],
    start = Date.now();
  let requests = 0,
    key;
  const audit = (x) =>
    appendFileSync(
      join(output, `${item.id}.events.jsonl`),
      JSON.stringify(x) + "\n",
      { mode: 0o600 },
    );
  session.agent.beforeToolCall = ({ toolCall, args }) => {
    if (toolCall.name === "read") {
      try {
        readAllowed(args.path, cwd);
      } catch {
        return {
          block: true,
          reason: "Read outside supplied inputs or missing file",
        };
      }
    }
  };
  session.agent.afterToolCall = ({ toolCall, args, result, isError }) => {
    const event = {
      name: toolCall.name,
      arguments: args,
      result: result.details ?? result,
      isError,
    };
    tools.push(event);
    audit({ kind: "tool", ...event });
  };
  const stream = session.agent.streamFunction;
  session.agent.streamFunction = (m, context, o) => {
    assert(!halted, "Campaign halted");
    assert(++requests <= 16, "Trial turn cap");
    key = admit(context, 4096, item.id);
    return stream(m, context, { ...o, ...options(4096, 0.7) });
  };
  session.subscribe((e) => {
    if (e.type === "message_end" && e.message.role === "assistant") {
      const m = e.message;
      complete(m, key);
      const event = {
        text: text(m),
        stopReason: m.stopReason,
        usage: m.usage,
        toolsRequested: m.content
          .filter((x) => x.type === "toolCall")
          .map((x) => ({ name: x.name, arguments: x.arguments })),
      };
      messages.push(event);
      audit({ kind: "assistant", ...event });
    }
  });
  try {
    await session.prompt(
      item.caseId === "mcp_canary"
        ? canaryPrompt
        : corpus.cases.find((c) => c.id === item.caseId).prompt,
    );
    const last = messages.at(-1),
      status =
        last?.stopReason === "stop" && last.text.trim()
          ? "executed"
          : budget.inflight[key]
            ? "provider_indeterminate"
            : halted
              ? "campaign_interrupted"
              : "bounded_failure";
    save(file, {
      ...item,
      status,
      final: last?.text ?? "",
      messages,
      tools,
      notebookCreated: existsSync(db),
      durationMs: Date.now() - start,
      systemPromptSha256: sha(session.agent.state.systemPrompt),
    });
    console.log(
      JSON.stringify({
        trial: item.id,
        phase: item.phase,
        case: item.caseId,
        status,
        requests,
        ms: Date.now() - start,
        reservedUSD: budget.reservedUSD,
        reportedUSD: budget.reportedUSD,
      }),
    );
    if (["provider_indeterminate", "campaign_interrupted"].includes(status)) {
      halted = true;
      throw new Error("Incomplete campaign; retain and reconcile");
    }
  } catch (error) {
    halted = true;
    throw error;
  } finally {
    session.dispose();
  }
}
async function score(item) {
  const target = join(output, `${item.id}.grade.json`);
  if (existsSync(target)) return;
  const result = load(join(output, `${item.id}.json`));
  assert(["executed", "bounded_failure"].includes(result.status));
  const rawPath = join(output, `${item.id}.judge.json`);
  assert(
    !existsSync(rawPath),
    "Ungraded judge response: reconcile, never resample",
  );
  const c = corpus.cases.find((x) => x.id === item.caseId);
  const context = {
    systemPrompt: judge,
    messages: [
      {
        role: "user",
        content: [
          {
            type: "text",
            text: JSON.stringify({
              prompt: c.prompt,
              criteria: c.expected_output.criteria,
              answer: result.final,
              execution: mcpFacts(result),
            }),
          },
        ],
        timestamp: Date.now(),
      },
    ],
  };
  const key = admit(context, 2048, `judge-${item.id}`),
    answer = await runtime.completeSimple(model, context, {
      ...options(2048, 0),
      reasoning: "off",
    });
  complete(answer, key);
  save(rawPath, {
    text: text(answer),
    stopReason: answer.stopReason,
    usage: answer.usage,
  });
  assert.equal(answer.stopReason, "stop");
  const value = JSON.parse(
    text(answer)
      .trim()
      .replace(/^```(?:json)?\s*/, "")
      .replace(/\s*```$/, ""),
  );
  save(target, {
    status: "advisory_same_model_judgment",
    ...executionChecks(gradeShape(c, value), result),
  });
  console.log(
    JSON.stringify({
      graded: item.id,
      reservedUSD: budget.reservedUSD,
      reportedUSD: budget.reportedUSD,
    }),
  );
}
async function batch(items, fn) {
  let next = 0;
  const errors = [];
  await Promise.all(
    Array.from({ length: 2 }, async () => {
      while (next < items.length && !halted) {
        const item = items[next++];
        try {
          await fn(item);
        } catch (e) {
          halted = true;
          errors.push(e);
        }
      }
    }),
  );
  if (errors.length) throw errors[0];
  assert(!halted && next === items.length, "Campaign incomplete");
}
